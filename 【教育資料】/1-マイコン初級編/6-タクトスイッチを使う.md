# 6 - タクトスイッチを使う

タクトスイッチと`digitalRead()`を使う
タクトスイッチを使ってLEDを光らせてみましょう
まずはタクトスイッチの構造を[このサイト](https://note.com/shingo_shirogane/n/n01f0484907a9)で勉強しましょう


  
  int ledPin = 13; // LEDを13番ピンに
  int inPin = 7;   // デジタルピン7にタクトスイッチ
  int val = 0;     // 読み取った値を保持する変数

  void setup() {
    Serial.begin(9600);        // シリアルポートを9600bpsで開く
    pinMode(ledPin, OUTPUT);   // LED用に出力に設定
    pinMode(inPin, INPUT);     // スイッチ用に入力に設定
  }

  void loop() {
    val = digitalRead(inPin);  // 入力ピンを読む
    Serial.println(val);       //読み取った値をシリアルモニターに表示
    digitalWrite(ledPin, val); // LEDのピンを読み取った値に変更
  }
